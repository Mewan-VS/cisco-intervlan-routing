! ===== SW2 (Cisco 2960) =====
! Paste in CLI tab after: enable -> configure terminal
hostname SW2
no ip domain-lookup
enable secret cisco123
!
vlan 10
 name HR
vlan 20
 name IT
vlan 30
 name FINANCE
vlan 99
 name NATIVE
vlan 999
 name PARKING
!
! --- Access ports ---
interface FastEthernet0/1
 description HR-PC2
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
interface FastEthernet0/2
 description IT-PC2
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
interface FastEthernet0/3
 description FIN-PC2
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
! Optional bonus server (IT VLAN)
interface FastEthernet0/4
 description IT-SERVER
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
! --- Trunk to SW1 ---
interface GigabitEthernet0/1
 description Trunk to SW1 G0/2
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,99
 switchport nonegotiate
!
! --- Security: park and shut unused ports ---
interface range FastEthernet0/5 - 24 , GigabitEthernet0/2
 switchport mode access
 switchport access vlan 999
 shutdown
!
line console 0
 password cisco
 login
!
end
write memory
